      const backgrounds=['--bg','--surface','--raised'].map(token);
      const values=Object.fromEntries(['--ink','--muted','--accent','--cyan','--violet','--green','--amber','--red'].map(k=>[k,Math.min(...backgrounds.map(bg=>ratio(token(k),bg)))]));
